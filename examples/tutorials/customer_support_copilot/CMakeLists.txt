cmake_minimum_required(VERSION 3.25)
project(customer_support_copilot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(a2a_cpp CONFIG REQUIRED)
add_library(support_tutorial STATIC src/tutorial.cpp src/model.cpp)
target_include_directories(support_tutorial PUBLIC include)
target_link_libraries(support_tutorial PUBLIC a2a::client a2a::server a2a::core a2a::http)
foreach(program support_client support_coordinator support_specialist)
  add_executable(${program} src/${program}.cpp)
  target_link_libraries(${program} PRIVATE support_tutorial)
endforeach()
