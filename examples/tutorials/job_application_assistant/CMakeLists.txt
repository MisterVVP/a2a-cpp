cmake_minimum_required(VERSION 3.25)
project(job_application_assistant LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(a2a_cpp CONFIG REQUIRED)

add_library(job_tutorial STATIC src/tutorial.cpp src/model.cpp)
target_include_directories(job_tutorial PUBLIC include)
target_link_libraries(job_tutorial PUBLIC a2a::client a2a::server a2a::core a2a::http)
foreach(program application_client application_coordinator profile_analyst)
  add_executable(${program} src/${program}.cpp)
  target_link_libraries(${program} PRIVATE job_tutorial)
endforeach()
